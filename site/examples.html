Examples

Five short files that cover the framework end to end: built-in robots, custom bodies, reward shaping, Gymnasium, and export.

Each example is a single file in examples/. Run it with Python, or hand it to the CLI.

Train a built-in robot

01

Train a built-in robot

examples/01_quickstart.py

The shortest path from install to a trained policy. Every built-in robot is a (robot, task) pair written with the same public API you use.

"""Train the built-in walker for 100 iterations, then report how far it gets.

    python examples/01_quickstart.py
"""
import os

import macgym as mg

ITERS = int(os.environ.get("MACGYM_ITERS", 100))
ENVS = int(os.environ.get("MACGYM_ENVS", 4096))

env = mg.make(*mg.robots.walker(), num_envs=ENVS)
result = mg.train(env, iters=ITERS, save="runs/quickstart_walker.npz")
print(f"final distance: {result.final.get('dist') or 0:.2f} m   policy: {result.policy}")
$ python examples/01_quickstart.py01_quickstart.py

Define your own robot

02

Define your own robot

examples/02_custom_robot.py

Capsules and hinge joints. The Metal kernel for this exact body plan is generated when you call mg.make.

"""A brand-new three-legged creature and its reward, in about 20 lines.

    python examples/02_custom_robot.py
"""
import os

import macgym as mg

robot = mg.Robot("tripod3")
torso = robot.body("torso", start=(-0.5, 0.7), end=(0.5, 0.7), radius=0.07)
for name, x in (("back", -0.5), ("mid", 0.0), ("front", 0.5)):
    thigh = robot.body(f"{name}_thigh", start=(x, 0.7), end=(x, 0.35), radius=0.04)
    shin = robot.body(f"{name}_shin", start=(x, 0.35), end=(x + 0.1, 0.05), radius=0.035)
    robot.joint(torso, thigh, at=(x, 0.7), limits=(-1.0, 1.0), motor=70, name=f"{name}_hip")
    robot.joint(thigh, shin, at=(x, 0.35), limits=(-1.5, 0.3), motor=50, name=f"{name}_knee")
robot.set_root(torso)

task = mg.Task(
    reward=lambda s: s.root.vx + 0.5 * (s.root.y > 0.45) - 0.002 * s.action_sq,  # go forward, stay up
    done=lambda s: s.root.y < 0.25,
    episode_steps=1000,
)

if __name__ == "__main__":
    env = mg.make(robot, task, num_envs=int(os.environ.get("MACGYM_ENVS", 4096)))
    mg.train(env, iters=int(os.environ.get("MACGYM_ITERS", 150)), save="runs/tripod3.npz")
$ macgym train examples/02_custom_robot.py02_custom_robot.py

Shape the reward

03

Shape the reward

examples/03_custom_reward.py

Rewards and terminations are lambdas over a batched State. Every field is an MLX array with one entry per robot.

"""Same cheetah body, new goal: hop BACKWARDS. Only the Task changes.

    python examples/03_custom_reward.py
"""
import os

import macgym as mg

robot, _ = mg.robots.cheetah()          # reuse the built-in body, throw away its task


def hop_backwards(s):
    airborne = (1 - s.contact("bfoot")) * (1 - s.contact("ffoot"))  # both feet off the ground
    return -s.root.vx + 0.5 * airborne - 0.1 * s.action_sq


task = mg.Task(reward=hop_backwards, episode_steps=1000)

if __name__ == "__main__":
    env = mg.make(robot, task, num_envs=int(os.environ.get("MACGYM_ENVS", 4096)))
    result = mg.train(env, iters=int(os.environ.get("MACGYM_ITERS", 150)), save="runs/cheetah_backflip.npz")
    print(f"distance (negative = backwards): {result.final.get('dist') or 0:.2f} m")
$ macgym train examples/03_custom_reward.py --envs 819203_custom_reward.py

Use it from Gymnasium

04

Use it from Gymnasium

examples/04_gymnasium.py

Wrap any MacGym env as a gymnasium.vector.VectorEnv (NumPy in, NumPy out) for Stable-Baselines3, CleanRL, or your own loop. Needs pip install -e ".[gym]".

"""Use a MacGym env from any Gymnasium-based library (SB3, CleanRL, ...).

to_gymnasium returns a gymnasium.vector.VectorEnv with numpy float32 in/out.
Autoreset is SAME_STEP: when an env finishes, the obs returned for it is already the
first obs of its next episode (see macgym/gym_compat.py).

    python examples/04_gymnasium.py
"""
import os

import numpy as np

import macgym as mg

N = int(os.environ.get("MACGYM_ENVS", 256))
STEPS = int(os.environ.get("MACGYM_STEPS", 500))

venv = mg.to_gymnasium(mg.make(*mg.robots.walker(), num_envs=N))
print("obs space:", venv.single_observation_space, " action space:", venv.single_action_space)

obs, info = venv.reset(seed=0)
episodes, returns = 0, []
for _ in range(STEPS):
    actions = venv.action_space.sample()                     # random policy
    obs, reward, terminated, truncated, info = venv.step(actions)
    done = terminated | truncated
    episodes += int(done.sum())
    returns += list(info["episode_return"][done])
assert obs.shape == (N, venv.single_observation_space.shape[0]) and obs.dtype == np.float32
print(f"{STEPS} steps x {N} envs, {episodes} episodes finished, "
      f"mean random-policy return {np.mean(returns) if returns else 0:.1f}")
venv.close()
$ python examples/04_gymnasium.py04_gymnasium.py

Export to CoreML

05

Export to CoreML

examples/05_export_coreml.py

Load a checkpoint and ship the policy. CoreML runs on the Neural Engine in a native app; ONNX goes anywhere. Needs pip install -e ".[export]".

"""Train briefly, then export the policy to CoreML (.mlpackage) and ONNX and run it outside MLX.

    python examples/05_export_coreml.py
"""
import os

import numpy as np

import macgym as mg

env = mg.make(*mg.robots.walker(), num_envs=int(os.environ.get("MACGYM_ENVS", 4096)))
result = mg.train(env, iters=int(os.environ.get("MACGYM_ITERS", 50)), save="runs/export_demo.npz")
os.makedirs("exports", exist_ok=True)

info = result.policy.export("exports/walker.onnx")
print(f"ONNX   -> {info['path']}  max |diff| vs MLX: {info['max_abs_err']:.1e}")
try:
    info = result.policy.export("exports/walker.mlpackage")
    err = info["max_abs_err"]
    print(f"CoreML -> {info['path']}  max |diff| vs MLX: " + (f"{err:.1e}" if err is not None else "n/a"))
    import coremltools as ct

    model = ct.models.MLModel(info["path"])
    obs = np.array(env.reset(), dtype=np.float32)[:1]
    print("CoreML action for one observation:", np.round(model.predict({"obs": obs})["action"][0], 3))
except ImportError as e:
    print("CoreML export skipped:", e)
$ macgym export runs/walker.npz --format coreml05_export_coreml.py
